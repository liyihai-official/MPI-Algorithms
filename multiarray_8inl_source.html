109template <size_t NumD>
110inline void multi_array_shape<NumD>::compute_strides()
111{
112 if constexpr (NumD == 0) return;
113
114 strides[NumD - 1] = 1;
115 for (size_t i = NumD - 1; i > 0; --i)
116 strides[i - 1] = strides[i] * dims[i];

228template <typename T, size_t NumD>
229template <std::integral... Exts>
230inline typename array<T, NumD>::reference
231array<T, NumD>::operator()(Exts... exts)
232{
233 static_assert(
234 sizeof...(exts) == NumD,
235 "Arguments count must match NumD.");
236
237 std::array<size_t, NumD> indices = {static_cast<size_t>(exts)...};
238
239 size_t flat_idx = 0;
240 for (size_t i = 0; i < NumD; ++i)
