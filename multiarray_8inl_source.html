125template <size_t NumD>
126inline void multi_array_shape<NumD>::compute_strides()
127{
128 if constexpr (NumD == 0) return;
129
130 strides[NumD - 1] = 1;
131 for (size_t i = NumD - 1; i > 0; --i)
132 strides[i - 1] = strides[i] * dims[i];
133}
