99template <typename T, size_t NumD>
100Cartesian<T, NumD>::Cartesian(
101 const array_shape& glob_shape,
102 MPI_Comm comm)
103 : global_shape(glob_shape),
104 local_shape(glob_shape)
105{
106 int orig_rank;
107 MPI_Comm_size(comm, &size);
108 MPI_Comm_rank(comm, &rank);
109
110 // Initializing
111 dims.fill(0);
112 periods.fill(1); // Turn On Periods of Boundaries
113 halos.fill(MPI_DATATYPE_NULL);
114 nbr_src.fill(0);
115 nbr_dest.fill(0);
116
117 // MPI_Topology
118 MPI_Dims_create(
119 size,
120 NumD,
121 dims.data());
122
123 MPI_Cart_create(
124 comm,
125 NumD,
126 dims.data(),
127 periods.data(),
128 0,
129 &comm_cart);
130
131 if (comm_cart != MPI_COMM_NULL)
132 {
133 MPI_Comm_rank(comm_cart, &rank);
134 MPI_Comm_size(comm_cart, &size);
135
136 MPI_Cart_coords(
137 comm_cart,
138 rank,
139 NumD,
140 coordinates.data());
141
146 auto Decomp = [](
147 const int n,
148 const int prob_size,
149 const int rank,
150 int& s,
151 int& e)
152 {
153 int n_loc{n / prob_size}, remain{n % prob_size};
154
155 s = rank * n_loc + 1;
156 s += ((rank < remain) ? rank : remain);
157
158 if (rank < remain) ++n_loc;
159 e = s + n_loc - 1;
160
161 if (e > n || rank == prob_size - 1) e = n;
162 return 0;
163 };
164
165 // Get neighbors and local shapes
166 for (size_t i = 0; i < NumD; ++i)
167 {
168 Decomp(
169 global_shape.dims[i],
170 dims[i],
171 coordinates[i],
172 starts[i],
173 ends[i]);
174
175 local_shape.dims[i] = ends[i] - starts[i] + 1 + 2; // Add halos
176 MPI_Cart_shift(
177 comm_cart,
178 i,
179 1,
180 &nbr_src[i],
181 &nbr_dest[i]);
182 }
183
184 local_shape.compute_strides();
185 } // end of if (comm_cart != MPI_COMM_NULL)
186}
