97template <typename T, size_t NumD>
98Cartesian<T, NumD>::Cartesian(
99 const array_shape& glob_shape,
100 MPI_Comm comm)
101 : global_shape(glob_shape),
102 local_shape(glob_shape)
103{
104 int size, orig_rank;
105 MPI_Comm_size(comm, &size);
106 MPI_Comm_rank(comm, &rank);
107
108 // Initializing
109 dims.fill(0);
110 periods.fill(1); // Turn On Periods of Boundaries
111 halos.fill(MPI_DATATYPE_NULL);
112 nbr_src.fill(0);

114
115 // MPI_Topology
116 MPI_Dims_create(size, NumD, dims.data());
117

119 comm,
120 NumD,
121 dims.data(),
122 periods.data(),
123 0,
124 &comm_cart);
125
126 if (comm_cart != MPI_COMM_NULL)
127 {
128 MPI_Comm_rank(comm_cart, &rank);

139 auto Decomp = [](
140 const int n,
141 const int prob_size,
142 const int rank,
143 int& s,
144 int& e)
145 {
146 int n_loc{n / prob_size}, remain{n % prob_size};
147
148 s = rank * n_loc + 1;
149 s += ((rank < remain) ? rank : remain);
150
151 if (rank < remain) ++n_loc;
152 e = s + n_loc - 1;
153
154 if (e > n || rank == prob_size - 1) e = n;
155 return 0;
156 };
157
158 // Get neighbors and local shapes
159 for (size_t i = 0; i < NumD; ++i)
160 {
161 Decomp(
162 global_shape.dims[i],
163 dims[i],
164 coordinates[i],
165 starts[i],
166 ends[i]);
167
168 local_shape.dims[i] = ends[i] - starts[i] + 1 + 2; // Add halos
169 MPI_Cart_shift(
170 comm_cart,
171 i,
172 1,
173 &nbr_src[i],
174 &nbr_dest[i]);
175 }
176
177 local_shape.compute_strides();
178 }
