28template <typename T, size_t NumD>
29void Bruck_Alltoall_noneblocking(
30 const multi_array::array<T, NumD>& send_arr,
31 multi_array::array<T, NumD>& recv_arr,
32 MPI_Comm comm)
33{
34 int rank, num_procs;
35 MPI_Comm_rank(comm, &rank);
36 MPI_Comm_size(comm, &num_procs);
37 MPI_Datatype base_type = mpi_type::mpi_type_traits<T>::get();
38
39 size_t total_elements{send_arr.size()};
40 size_t elements_per_proc{total_elements / num_procs};
41
42 const T* send_data{send_arr.data()};
43 T* recv_data{recv_arr.data()};
44
45 // Local data rotation
46 for (int j = 0; j < num_procs; ++j)
47 {
48 int dest_rank{static_cast<int>((j - rank + num_procs) % num_procs)};
49 std::memcpy(
50 &recv_data[dest_rank * elements_per_proc],
51 &send_data[j * elements_per_proc],
52 elements_per_proc * sizeof(T));
53 }
54
55 // None blocking communication
56 std::vector<T> recv_temp(total_elements);
57 int num_steps{static_cast<int>(std::ceil(std::log2(num_procs)))};
58
59 for (int step = 0; step < num_steps; ++step)
60 {
61 int distance{1 << step}; // bitwise left shift to calculate 2^step
62 int sent_to{(rank + distance) % num_procs};
63 int recv_from{(rank - distance + num_procs) % num_procs};
64
65 // calculate the displacements
66 std::vector<int> displacement;
67 for (int j = 0; j < num_procs; ++j)
68 {
69 if ((j >> step) & 1)
70 {
71 displacement.push_back(j * elements_per_proc);
72 }
73 }
74
75 int num_blocks{static_cast<int>(displacement.size())};
76 if (num_blocks == 0) continue; // no need for sending.
77
78 // MPI_Datatype
79 MPI_Datatype step_type;
80 MPI_Type_create_indexed_block(
81 num_blocks,
82 elements_per_proc,
83 displacement.data(),
84 base_type,
85 &step_type);
86
87 MPI_Type_commit(&step_type);
88
89 // Non-blocking send and receive
90 MPI_Request reqs[2];
91
92 MPI_Isend(
93 recv_data,
94 1,
95 step_type,
96 sent_to,
97 0,
98 comm,
99 &reqs[0]);
100
101 MPI_Irecv(
102 recv_temp.data(),
103 num_blocks * elements_per_proc,
104 base_type,
105 recv_from,
106 0,
107 comm,
108 &reqs[1]);
109
110 MPI_Waitall(2, reqs, MPI_STATUSES_IGNORE);
111
112 int position{0};
113 int insize_bytes{static_cast<int>(num_blocks * elements_per_proc * sizeof(T))};
114 MPI_Unpack(
115 recv_temp.data(),
116 insize_bytes,
117 &position,
118 recv_data,
119 1,
120 step_type,
121 comm);
122 MPI_Type_free(&step_type);
123 }
124
125 // Final data rotation
126 std::vector<T> final_temp(total_elements);
127 for (int j = 0; j < num_procs; ++j)
128 {
129 int orig_block{(num_procs + rank - j) % num_procs};
130 std::memcpy(
131 &final_temp[orig_block * elements_per_proc],
132 &recv_data[j * elements_per_proc],
133 elements_per_proc * sizeof(T));
134 }
135 std::memcpy(
136 recv_data,
137 final_temp.data(),
138 total_elements * sizeof(T));
139}
