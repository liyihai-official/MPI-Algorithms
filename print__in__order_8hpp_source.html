23inline void print_in_order(
24 const std::string& local_text,
25 MPI_Comm comm,
26 const std::string& title = "")
27{
28 int rank{-1}, num_procs{0};
29 MPI_Comm_rank(comm, &rank);
30 MPI_Comm_size(comm, &num_procs);
31
32 if (local_text.size() > static_cast<size_t>(INT_MAX))
33 {
34 std::cerr << "print_in_order: local text exceeds INT_MAX bytes.\n";
35 MPI_Abort(comm, 1);
36 return;
37 }
38
39 int local_size{static_cast<int>(local_text.size())};
40 std::vector<int> sizes(rank == 0 ? num_procs : 0);
41 std::vector<int> displs(rank == 0 ? num_procs : 0);
42
43 MPI_Gather(
44 &local_size,
45 1,
46 MPI_INT,
47 sizes.data(),
48 1,
49 MPI_INT,
50 0,
51 comm);
52
53 int total_size{0};
54 if (rank == 0)
55 {
56 for (int p = 0; p < num_procs; ++p)
57 {
58 if (sizes[p] > INT_MAX - total_size)
59 {
60 std::cerr << "print_in_order: total text exceeds INT_MAX bytes.\n";
61 MPI_Abort(comm, 1);
62 return;
63 }
64 displs[p] = total_size;
65 total_size += sizes[p];
66 }
67 }
68
69 // Keep a valid root buffer even when every process supplies empty text.
70 std::vector<char> text(rank == 0 ? (total_size > 0 ? total_size : 1) : 0);
71 MPI_Gatherv(
72 local_text.data(), local_size, MPI_CHAR, text.data(), sizes.data(), displs.data(), MPI_CHAR, 0, comm);
73
74 if (rank == 0)
75 {
76 if (!title.empty()) std::cout << "\n========== " << title << " ==========\n";
77 for (int p = 0; p < num_procs; ++p)
78 {
79 std::cout << "\n[rank " << p << "]\n";
80 std::cout.write(text.data() + displs[p], sizes[p]);
81 if (sizes[p] == 0 || text[displs[p] + sizes[p] - 1] != '\n')
82 std::cout << '\n';
83 }
84 std::cout.flush();
85 }
86 MPI_Barrier(comm);
87}
